<script>
  (function() {
    // ---------- Daily step commitment logic ----------
    const commitBtn = document.getElementById('commitBtn');
    const stepInput = document.getElementById('stepInput');
    const stepFeedback = document.getElementById('stepFeedback');
    
    // Load saved daily commitment from localStorage
    function loadDailyCommitment() {
      const savedCommit = localStorage.getItem('skillmax_daily_commit');
      const savedDate = localStorage.getItem('skillmax_commit_date');
      const today = new Date().toDateString();
      if (savedCommit && savedDate === today) {
        stepFeedback.innerHTML = `✅ You committed today: "${savedCommit}". Keep crushing it! Come back tomorrow for a new step.`;
        stepInput.disabled = true;
        commitBtn.disabled = true;
        commitBtn.style.opacity = '0.6';
        commitBtn.style.cursor = 'default';
      } else if (savedCommit && savedDate !== today) {
        // new day, enable fresh commitment
        stepInput.disabled = false;
        commitBtn.disabled = false;
        commitBtn.style.opacity = '1';
        stepFeedback.innerHTML = `🌅 New day! Ready for your next micro-action. Yesterday you did: "${savedCommit}". Set today's step.`;
        stepInput.value = "";
      } else {
        stepInput.disabled = false;
        commitBtn.disabled = false;
        commitBtn.style.opacity = '1';
        stepFeedback.innerHTML = `💡 Click commit to lock in your daily action. Progress over perfection.`;
      }
    }
    
    function saveDailyCommit(commitText) {
      if (!commitText.trim()) {
        stepFeedback.innerHTML = `⚠️ Please write a small action step before committing.`;
        return false;
      }
      const today = new Date().toDateString();
      localStorage.setItem('skillmax_daily_commit', commitText.trim());
      localStorage.setItem('skillmax_commit_date', today);
      stepFeedback.innerHTML = `🎉 Amazing! Today's step: "${commitText.trim()}". Small win → big momentum. Come back tomorrow for your next upgrade.`;
      stepInput.disabled = true;
      commitBtn.disabled = true;
      commitBtn.style.opacity = '0.6';
      return true;
    }
    
    commitBtn.addEventListener('click', () => {
      const commitValue = stepInput.value;
      saveDailyCommit(commitValue);
    });
    
    // ---------- Streak Tracker (simple counters) ----------
    // pillars: gen, comm, health, finance
    let streaks = {
      gen: 0,
      comm: 0,
      health: 0,
      finance: 0
    };
    
    // Load saved streaks from localStorage
    function loadStreaks() {
      const saved = localStorage.getItem('skillmax_streaks');
      if (saved) {
        try {
          const parsed = JSON.parse(saved);
          if (parsed.gen !== undefined) streaks = parsed;
        } catch(e) {}
      }
      updateStreakUI();
    }
    
    function updateStreakUI() {
      document.getElementById('genKnowledgeCount').innerText = streaks.gen;
      document.getElementById('commCount').innerText = streaks.comm;
      document.getElementById('healthCount').innerText = streaks.health;
      document.getElementById('financeCount').innerText = streaks.finance;
      localStorage.setItem('skillmax_streaks', JSON.stringify(streaks));
    }
    
    function incrementStreak(pillar) {
      if (pillar === 'gen') {
        streaks.gen += 1;
      } else if (pillar === 'comm') {
        streaks.comm += 1;
      } else if (pillar === 'health') {
        streaks.health += 1;
      } else if (pillar === 'finance') {
        streaks.finance += 1;
      }
      updateStreakUI();
      // Optional micro-feedback effect - small notif in console but we can also show a temporary tooltip
      const msg = `➕ +1 day to ${getPillarName(pillar)}! Keep the streak alive.`;
      showTemporaryToast(msg);
    }
    
    function getPillarName(pillar) {
      const map = { gen: "General Knowledge", comm: "Communication", health: "Health", finance: "Finance" };
      return map[pillar];
    }
    
    function showTemporaryToast(message) {
      // create a quick floating message near bottom right, disappears after 2s
      const toast = document.createElement('div');
      toast.innerText = message;
      toast.style.position = 'fixed';
      toast.style.bottom = '24px';
      toast.style.right = '24px';
      toast.style.backgroundColor = '#1e6f5c';
      toast.style.color = 'white';
      toast.style.padding = '12px 20px';
      toast.style.borderRadius = '60px';
      toast.style.fontWeight = '500';
      toast.style.fontSize = '0.9rem';
      toast.style.boxShadow = '0 6px 16px rgba(0,0,0,0.1)';
      toast.style.zIndex = '999';
      toast.style.opacity = '0';
      toast.style.transition = 'opacity 0.2s';
      document.body.appendChild(toast);
      setTimeout(() => { toast.style.opacity = '1'; }, 10);
      setTimeout(() => {
        toast.style.opacity = '0';
        setTimeout(() => { if(toast.parentNode) toast.remove(); }, 300);
      }, 2000);
    }
    
    function resetAllStreaks() {
      streaks = { gen: 0, comm: 0, health: 0, finance: 0 };
      updateStreakUI();
      showTemporaryToast("🔄 All skill streaks have been reset. New beginning!");
    }
    
    // attach event listeners to all streak buttons
    const streakBtns = document.querySelectorAll('.streak-btn');
    streakBtns.forEach(btn => {
      btn.addEventListener('click', (e) => {
        const pillar = btn.getAttribute('data-pillar');
        if (pillar) incrementStreak(pillar);
      });
    });
    
    const resetBtn = document.getElementById('resetTrackerBtn');
    if (resetBtn) resetBtn.addEventListener('click', resetAllStreaks);
    
    // initial load
    loadDailyCommitment();
    loadStreaks();
    
    // Re-check date when page becomes visible (in case user keeps tab open overnight)
    window.addEventListener('focus', () => {
      // re-evaluate daily commitment locking based on date
      const savedDate = localStorage.getItem('skillmax_commit_date');
      const today = new Date().toDateString();
      if (savedDate !== today) {
        // enable new day
        const savedCommit = localStorage.getItem('skillmax_daily_commit');
        stepInput.disabled = false;
        commitBtn.disabled = false;
        commitBtn.style.opacity = '1';
        if (savedCommit) {
          stepFeedback.innerHTML = `🌞 New day! Yesterday's step: "${savedCommit}". What's today's micro-action?`;
        } else {
          stepFeedback.innerHTML = `💡 New day — set your daily step and commit to leveling up.`;
        }
        stepInput.value = "";
      } else {
        // already committed today: ensure disabled state
        const savedCommit = localStorage.getItem('skillmax_daily_commit');
        if (savedCommit) {
          stepInput.disabled = true;
          commitBtn.disabled = true;
          commitBtn.style.opacity = '0.6';
          stepFeedback.innerHTML = `✅ You committed today: "${savedCommit}". Keep going! Come back tomorrow.`;
        }
      }
    });
  })();
</script>
